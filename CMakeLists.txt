cmake_minimum_required(VERSION 3.20)
project(NewCheatEngine VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Enable compiler warnings & optimizations
if (MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -O2)
endif()

# Shared VEH agent DLL must NOT inherit global -static flag on MinGW
if (WIN32)
    add_subdirectory(veh_agent)
endif()

if (MINGW)
    add_link_options(-static -static-libgcc -static-libstdc++)
endif()

# Subdirectories
add_subdirectory(third_party/lua)
add_subdirectory(engine)
add_subdirectory(target_dummy)
add_subdirectory(cli)
add_subdirectory(server)
option(NCE_BUILD_GUI "Build the Dear ImGui GUI application" ON)
if (NCE_BUILD_GUI)
    if (WIN32)
        add_subdirectory(gui)
    else()
        find_package(SDL2 QUIET)
        find_package(OpenGL QUIET)
        if (NOT SDL2_FOUND)
            find_package(PkgConfig QUIET)
            if (PkgConfig_FOUND)
                pkg_check_modules(SDL2_PKG QUIET sdl2)
            endif()
        endif()
        if ((SDL2_FOUND OR SDL2_PKG_FOUND) AND OpenGL_FOUND)
            add_subdirectory(gui)
        else()
            message(STATUS "Skipping nce_gui: SDL2 and/or OpenGL dev libraries not found (install libsdl2-dev libgl-dev on Debian/Ubuntu).")
        endif()
    endif()
endif()

option(NCE_BUILD_TESTS "Build the engine unit tests" ON)
if (NCE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
